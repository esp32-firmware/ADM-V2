  <script>
    let masterStudentList = [];
    let selectedStudents = new Set();
    let isAllSelected = false;

    // --- Clock Logic ---
    setInterval(() => {
      const now = new Date();
      const options = { weekday: 'short', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit', second: '2-digit' };
      document.getElementById('live-clock').innerText = now.toLocaleString('en-IN', options);
    }, 1000);

    // --- Boot Load ---
    window.onload = function() {
      document.getElementById('loader').style.display = 'block';
      google.script.run.withSuccessHandler(renderStudents).getWebAppStudents();
    };

    function renderStudents(students) {
      masterStudentList = students;
      selectedStudents.clear();

      document.getElementById('loader').style.display = 'none';
      const container = document.getElementById('student-list');
      container.innerHTML = '';

      students.forEach((student, index) => {
        const row = document.createElement('div');
        row.className = 'student-row';
        row.id = `row-${index}`;
        row.dataset.name = student.name.toLowerCase();
        
        if (student.isPresent) {
          row.classList.add('selected');
          selectedStudents.add(index);
        }

        row.innerHTML = `
          <div class="student-info">
            <span class="student-name">${student.name}</span>
            <span class="student-reg">ID: ${student.regNo || student.uid}</span>
          </div>
          <div class="custom-checkbox">
            <svg class="check-icon"><use href="#icon-check"></use></svg>
          </div>
        `;

        row.onclick = () => toggleStudent(index);
        container.appendChild(row);
      });

      updateUploadButton();
    }

    // --- Batch Selection Logic ---
    function toggleStudent(index) {
      const row = document.getElementById(`row-${index}`);
      if (selectedStudents.has(index)) {
        selectedStudents.delete(index);
        row.classList.remove('selected');
      } else {
        selectedStudents.add(index);
        row.classList.add('selected');
      }
      updateUploadButton();
    }

    function toggleSelectAll() {
      const visibleRows = document.querySelectorAll('.student-row:not(.hidden)');
      isAllSelected = !isAllSelected;
      
      visibleRows.forEach(row => {
        const index = parseInt(row.id.split('-')[1]);
        if (isAllSelected) {
          selectedStudents.add(index);
          row.classList.add('selected');
        } else {
          selectedStudents.delete(index);
          row.classList.remove('selected');
        }
      });
      
      document.getElementById('selectAllBtn').innerText = isAllSelected ? "Clear All" : "Select All";
      updateUploadButton();
    }

    function updateUploadButton() {
      const btn = document.getElementById('uploadBtn');
      const count = selectedStudents.size;
      btn.innerText = `Sync Database (${count})`;
      btn.disabled = false;
    }

    // --- Search Logic (Starts-With Matching) ---
    function filterStudents() {
      const query = document.getElementById('searchInput').value.toLowerCase().trim();
      const rows = document.querySelectorAll('.student-row');
      
      rows.forEach(row => {
        if (!query) {
          row.classList.remove('hidden');
          return;
        }

        const fullName = row.dataset.name;
        const nameParts = fullName.split(' ');
        
        const matchFound = fullName.startsWith(query) || nameParts.some(part => part.startsWith(query));

        if (matchFound) {
          row.classList.remove('hidden');
        } else {
          row.classList.add('hidden');
        }
      });
    }

    // --- NEW Custom Sync Confirmation Logic ---
    function openSyncModal() {
      const textElement = document.getElementById('syncModalText');
      textElement.innerText = `You are about to sync the database. This will update today's sheet to match the ${selectedStudents.size} ticked students.`;
      document.getElementById('syncModal').style.display = 'flex';
    }

    function closeSyncModal() {
      document.getElementById('syncModal').style.display = 'none';
    }

    function executeUpload() {
      closeSyncModal(); // Hide the modal immediately

      const payload = Array.from(selectedStudents).map(index => masterStudentList[index]);
      const btn = document.getElementById('uploadBtn');
      btn.innerText = "Syncing...";
      btn.disabled = true;

      google.script.run
        .withSuccessHandler((response) => {
          showToast(response.message);
          btn.innerText = `Sync Database (${selectedStudents.size})`;
          btn.disabled = false;
        })
        .withFailureHandler((err) => {
          showToast("Error: " + err.message);
          btn.innerText = `Sync Database (${selectedStudents.size})`;
          btn.disabled = false;
        })
        .submitWebAppAttendance(payload);
    }

    // --- Guest Logic ---
    function openGuestModal() {
      document.getElementById('guestModal').style.display = 'flex';
      document.getElementById('guestNameInput').focus();
    }

    function closeGuestModal() {
      document.getElementById('guestModal').style.display = 'none';
      document.getElementById('guestNameInput').value = '';
    }

    function submitGuest() {
      const nameInput = document.getElementById('guestNameInput').value.trim();
      if (!nameInput) {
        showToast("Please enter a name first.");
        return;
      }

      const btn = document.getElementById('guestSubmitBtn');
      btn.innerText = "Adding...";
      btn.disabled = true;

      google.script.run
        .withSuccessHandler((response) => {
          showToast(response.message);
          btn.innerText = "Add Guest";
          btn.disabled = false;
          closeGuestModal();
        })
        .withFailureHandler((err) => {
          showToast("Error: " + err.message);
          btn.innerText = "Add Guest";
          btn.disabled = false;
        })
        .submitGuestAttendance(nameInput);
    }

    // --- Utilities ---
    function showToast(msg) {
      const toast = document.getElementById('toast');
      toast.innerText = msg;
      toast.classList.add('show');
      setTimeout(() => toast.classList.remove('show'), 4000);
    }
  </script>
